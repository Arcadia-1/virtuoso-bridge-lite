create_rc.py · Python 
from virtuoso_bridge import VirtuosoClient
from virtuoso_bridge.virtuoso.schematic.ops import (
    schematic_create_inst_by_master_name as inst,
    schematic_create_wire_between_instance_terms as wire,
)

client = VirtuosoClient.from_env()
# Choose an existing library and a new cell name.
with client.schematic.create("my_lib", "rc_demo") as sch:
    sch.add(inst("analogLib", "res", "symbol",
                 "R0", 0.5, 0.0, "R90"))
    sch.add(inst("analogLib", "cap", "symbol",
                 "C0", 1.5, -0.5, "R0"))
    sch.add(wire("R0", "MINUS", "C0", "PLUS"))
# The context manager checks and saves the cellview.

simulate.py · Python 
from pathlib import Path
from virtuoso_bridge.spectre import SpectreSimulator
from virtuoso_bridge.spectre.psf import (
    frequency_hz, read_psf_ascii, result_file, vector,
)

sim = SpectreSimulator.from_env(work_dir="output")
result = sim.run_simulation(Path("tb.scs"))
if not result.ok:
    raise RuntimeError(result.errors)

raw_psf = Path(result.metadata["output_dir"])
ac = read_psf_ascii(result_file(raw_psf, "*.ac"))
frequency = frequency_hz(ac)
vout = vector(ac, "VOUT")
# Use your own AC testbench and output signal.

persistent_job.py · Python 
from virtuoso_bridge import VirtuosoClient
from virtuoso_bridge.virtuoso.maestro.jobs import (
    MaestroJobManager,
)

client = VirtuosoClient.from_env()
# Open an existing, configured Maestro cellview.
session = client.maestro.open_gui_session("my_lib", "my_tb")
jobs = MaestroJobManager.from_client(client)
job = jobs.submit(session=session)

# Persist this ID for a later Python process.
run_id = job.run_id
restored = jobs.load(run_id)
status = jobs.status(restored)
# Observation does not resubmit the simulation.

02
Keep the durable run handle保留持久化任务标识

Observe the original run继续观察原有任务

An uncertain submission is reconciled, not blindly repeated.提交状态不明时先核对原任务，不盲目重复提交。

Persistent handles bind observations to the original session and run. For a genuinely local GUI, explicitly use local_gui=True.通过持久化标识绑定原始会话与运行记录。真正的本机 GUI 场景需显式使用 local_gui=True。

QUICK START / macOS · Linux 
# 1. Install from source
git clone https://github.com/Arcadia-1/virtuoso-bridge-lite.git
cd virtuoso-bridge-lite
uv venv .venv
source .venv/bin/activate
uv pip install -e .

# 2. Connect to your EDA host
virtuoso-bridge init user@host
virtuoso-bridge start

Next: paste the exact load(...) line printed by start into your chosen Virtuoso CIW. Then run virtuoso-bridge status and virtuoso-bridge eval '1+2'.接下来：将 start 输出的完整 load(...) 命令粘贴到选定的 Virtuoso CIW，然后运行 virtuoso-bridge status 与 virtuoso-bridge eval '1+2'。
